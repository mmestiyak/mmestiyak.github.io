{{- $r := .res -}} {{- $w := .width | default 1400 -}} {{- $url := $r.RelPermalink -}} {{- if eq $r.ResourceType "image" -}} {{- $sub := $r.MediaType.SubType -}} {{- if and (ne $sub "svg") (ne $sub "gif") -}} {{- $img := $r -}} {{- if gt $r.Width $w -}}{{- $img = $r.Resize (printf "%dx webp q82" $w) -}} {{- else -}}{{- $img = $r.Resize (printf "%dx webp q82" $r.Width) -}}{{- end -}} {{- $url = $img.RelPermalink -}} {{- end -}} {{- end -}} {{- return $url -}}
